from data_utils import connect_to_db, disconnect_from_db
from psycopg2.extras import execute_batch
import logging
from datetime import datetime

logger = logging.getLogger(__name__)

def weather_data_clean(table_name):
    conn, cur = None, None
    try:
        conn, cur = connect_to_db()
        cur.execute(f"""
             INSERT INTO silver.weather_clean (
                station_id,
                date,
                temp_c,
                precipitation_mm,
                snow,
                wind_dir,
                wind_speed
            )
            SELECT
                substring(source FROM '(D?\\d+)$') AS station_id,
                (payload::json ->> 'time')::timestamp::date AS date,
                round(avg((payload::json ->> 'temp')::numeric), 2) AS temp_c,
                round(avg((payload::json ->> 'prcp')::numeric), 2) AS precipitation_mm,
                round(avg((payload::json ->> 'snow')::numeric), 2) AS snow,
                round(avg((payload::json ->> 'wdir')::numeric), 2) AS wind_dir,
                round(avg((payload::json ->> 'wspd')::numeric), 2) AS wind_speed
            FROM bronze.{table_name}
            GROUP BY substring(source FROM '(D?\\d+)$'), (payload::json ->> 'time')::timestamp::date
            ON CONFLICT (station_id, date) DO NOTHING;
        """)
        conn.commit()

    except Exception as e:
        logger.error(f"Error Inserting into database: {e}")
    finally:
        if conn and cur:
            disconnect_from_db(conn, cur)

weather_data_clean("meteostat_raw")
